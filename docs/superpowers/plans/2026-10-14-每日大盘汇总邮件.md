# 每日 A股大盘汇总邮件 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 每天收盘数据齐备后，自动汇总 A股大盘 ①-⑥ 六大板块，渲染为 HTML 邮件（红涨绿跌）发送到指定邮箱，并归档 HTML、记录 `task_run`。

**Architecture:** 沿用现有「APScheduler + `task_run` + TaskService 串行队列」体系，新增 `kind=report` 任务。新增指数日线数据源（baostock `query_history_k_data_plus`）落独立目录 `index_daily/`；其余板块由现有个股 Parquet 经 DuckDB 跨股聚合。计算层为纯函数（DataFrame → 指标 dict），渲染层 Jinja2，发送层 smtplib。触发前先做「数据就绪检查」，未就绪则 `skipped` 不发信。

**Tech Stack:** Python 3.10+ / FastAPI / SQLAlchemy(SQLite) / APScheduler / pandas / pyarrow / **duckdb** / **Jinja2** / smtplib / Vue3(原生 fetch)

**权威 spec：** `docs/superpowers/specs/2026-10-13-每日大盘汇总邮件-design.md`

```python
# -*- coding: utf-8 -*-
"""指数 Parquet 存储：按指数代码分文件，写/读/增量 merge/最新日。

与个股 parquet_store 分开：指数无复权/估值，仅约 10 列；
不进个股数据集扫描与 stock_freshness。
"""
from pathlib import Path

import pandas as pd

INDEX_COLS = ["date", "code", "open", "high", "low", "close",
              "preclose", "volume", "amount", "pctChg"]
NUMERIC_COLS = ["open", "high", "low", "close", "preclose", "volume", "amount", "pctChg"]


def normalize(code: str) -> str:
    """sh.000001 → sh000001（文件名不含点，与个股命名一致）。"""
    return str(code).replace(".", "").strip()


def index_path(root, code: str) -> Path:
    return Path(root) / f"{normalize(code)}.parquet"


def _cast(df: pd.DataFrame) -> pd.DataFrame:
    """数值列强制转数值，缺失列补空，裁到 INDEX_COLS 顺序。"""
    df = df.copy()
    for c in NUMERIC_COLS:
        if c in df.columns:
            df[c] = pd.to_numeric(df[c], errors="coerce")
    for c in INDEX_COLS:
        if c not in df.columns:
            df[c] = None
    if "date" in df.columns:
        df["date"] = df["date"].astype(str)
    return df[INDEX_COLS]


def write_index(root, code: str, df: pd.DataFrame) -> Path:
    """全量新建：按日期升序去重后落 {code}.parquet。"""
    df = _cast(df)
    df = df.sort_values("date").drop_duplicates("date", keep="last")
    p = index_path(root, code)
    p.parent.mkdir(parents=True, exist_ok=True)
    df.to_parquet(p, index=False)
    return p


def read_index(root, code: str) -> pd.DataFrame:
    p = index_path(root, code)
    if not p.exists():
        return pd.DataFrame(columns=INDEX_COLS)
    return pd.read_parquet(p)


def merge_index(root, code: str, df: pd.DataFrame) -> pd.DataFrame:
    """增量合并：读旧 → concat 新 → 按日期去重升序 → 重写。"""
    base = read_index(root, code)
    new = _cast(df)
    merged = new if base.empty else pd.concat([_cast(base), new], ignore_index=True)
    merged = merged.sort_values("date").drop_duplicates("date", keep="last")
    write_index(root, code, merged)
    return merged


def latest_date(root, code: str):
    """返回已入库最新日期字符串，无文件/空文件返回 None。"""
    p = index_path(root, code)
    if not p.exists():
        return None
    df = pd.read_parquet(p, columns=["date"])
    if df.empty:
        return None
    return str(df["date"].max())


def list_codes(root) -> list:
    """列出目录内已有指数代码（文件名去点）。"""
    d = Path(root)
    if not d.is_dir():
        return []
    return sorted(p.stem for p in d.glob("*.parquet"))
```

```python
# -*- coding: utf-8 -*-
"""指数日线抓取：按指数代码区间拉取，落 {index_dir}/{code}.parquet；取不到则跳过。

- 每股起点 = 该指数已入库最新日的次一日，无则 start_default（默认 2005-01-01）。
- 单指数查询空结果/异常 → 记 skipped，不写文件、不报错（不阻断）。
- 数据源通过模块级钩子 _fetch_index 注入：默认空实现（便于测试替换），
  运行时由接线层 bind_source 绑到真实 BaostockDataSource.index_daily。
"""
import logging
from datetime import date, timedelta
from pathlib import Path

import pandas as pd

from . import index_store

logger = logging.getLogger("app.services.index_ingest")

_fetch_index = lambda code, start, end: []   # noqa: E731


def bind_source(ds) -> None:
    """绑定真实数据源的指数日线方法。"""
    global _fetch_index
    _fetch_index = ds.index_daily


def _next_day(iso: str) -> str:
    return (date.fromisoformat(iso) + timedelta(days=1)).isoformat()


def run_index_fetch(root, codes, end_date: str, start_default: str = "2005-01-01",
                    log=None) -> dict:
    """对 codes 逐个增量抓取至 end_date，写 {root}/{code}.parquet。

    返回 {"fetched": {code: rows}, "skipped": [code], "latest": {code: date}}。
    """
    def _log(m: str) -> None:
        if log:
            log(m)
        logger.info(m)

    root = Path(root)
    fetched, skipped, latest = {}, [], {}
    for code in codes:
        last = index_store.latest_date(root, code)
        start = _next_day(last) if last else start_default
        if start > end_date:
            latest[code] = last
            _log(f"指数 {code}: 已最新({last})，跳过")
            continue
        try:
            rows = _fetch_index(code, start, end_date)
        except Exception as exc:  # noqa: BLE001
            skipped.append(code)
            _log(f"指数 {code}: 抓取异常，跳过（{exc}）")
            continue
        if not rows:
            skipped.append(code)
            _log(f"指数 {code}: 无数据，跳过")
            continue
        index_store.merge_index(root, code, pd.DataFrame(rows))
        fetched[code] = len(rows)
        latest[code] = index_store.latest_date(root, code)
        _log(f"指数 {code}: 写入 {len(rows)} 行，最新 {latest[code]}")
    return {"fetched": fetched, "skipped": skipped, "latest": latest}
```

```python
# -*- coding: utf-8 -*-
"""大盘汇总指标计算。

分两层：
- 纯函数层（本模块 compute_* / limit_threshold）：输入 DataFrame → 指标 dict，便于单测；
- IO 层（build_summary）：用 DuckDB 跨股读 parquet + index_store 读指数，拼装完整指标。

原则：数据缺失返回 None / 空 → 渲染层显示「数据缺失」，不造假。
"""
import math
from pathlib import Path

import pandas as pd

# 涨跌停幅度（近似口径，单位 %）
MAIN_BOARD = 10.0
STAR_GEM = 20.0
ST = 5.0
BEIJING = 30.0
TOL = 0.1   # 阈值容差：pctChg 达到 (阈值 ± 容差) 即判定

INDEX_NAMES = {
    "sh000001": "上证综指", "sz399001": "深证成指", "sz399006": "创业板指",
    "sh000688": "科创50", "sh000016": "上证50", "sh000300": "沪深300",
    "sh000905": "中证500", "sh000852": "中证1000",
}


def _truthy(v) -> bool:
    return str(v).strip().lower() in ("1", "1.0", "true", "yes")


def limit_threshold(code: str, is_st: bool) -> float:
    """涨跌停幅度（近似）：ST 5%，北交所 30%，科创/创业板 20%，其余 10%。"""
    if is_st:
        return ST
    c = str(code)
    market, num = (c[:2], c[2:]) if c[:2] in ("sh", "sz", "bj") else ("", c)
    if market == "bj" or num.startswith(("4", "8")):
        return BEIJING
    if num.startswith(("688", "689", "300", "301")):
        return STAR_GEM
    return MAIN_BOARD


def compute_breadth(df: pd.DataFrame) -> dict:
    """涨跌/平盘/涨停/跌停/停牌家数（剔停牌后统计涨跌）。"""
    cols = {"up": None, "down": None, "flat": None,
            "limit_up": None, "limit_down": None, "suspended": None, "total": 0}
    if df is None or df.empty:
        return cols

    st_num = pd.to_numeric(df.get("tradestatus"), errors="coerce")
    vol = pd.to_numeric(df.get("volume"), errors="coerce").fillna(0)
    suspended_mask = (st_num != 1) | (vol == 0)
    live = df[~suspended_mask]
    pct = pd.to_numeric(live["pctChg"], errors="coerce")

    limit_up = limit_down = 0
    for code, p, isst in zip(live["code"], pct, live.get("isST", pd.Series([0] * len(live)))):
        if p is None or (isinstance(p, float) and math.isnan(p)):
            continue
        th = limit_threshold(code, _truthy(isst))
        if p >= th - TOL:
            limit_up += 1
        elif p <= -(th - TOL):
            limit_down += 1

    return {
        "up": int((pct > 0).sum()),
        "down": int((pct < 0).sum()),
        "flat": int((pct == 0).sum()),
        "limit_up": int(limit_up),
        "limit_down": int(limit_down),
        "suspended": int(suspended_mask.sum()),
        "total": int(len(live)),
    }
```

```python
def _num_sum(df: pd.DataFrame, col: str):
    if df is None or df.empty or col not in df.columns:
        return None
    s = pd.to_numeric(df[col], errors="coerce")
    return None if s.dropna().empty else float(s.sum())


def compute_volume(df: pd.DataFrame, prev_amount, ma5_amount) -> dict:
    """全市场成交额/量、平均换手率；对比前一日与近 5 日均值给出放量/缩量。"""
    if df is None or df.empty:
        return {"amount": None, "volume": None, "turn_avg": None,
                "vs_prev_pct": None, "vs_ma5_pct": None, "state": None}
    amount = _num_sum(df, "amount")
    volume = _num_sum(df, "volume")
    turn = pd.to_numeric(df.get("turn"), errors="coerce")
    turn_avg = None if turn.dropna().empty else float(turn.mean())

    vs_prev = (amount / prev_amount - 1) * 100 if (amount is not None and prev_amount) else None
    vs_ma5 = (amount / ma5_amount - 1) * 100 if (amount is not None and ma5_amount) else None
    state = None
    if vs_ma5 is not None:
        state = "放量" if vs_ma5 > 0 else "缩量"
    return {"amount": amount, "volume": volume, "turn_avg": turn_avg,
            "vs_prev_pct": vs_prev, "vs_ma5_pct": vs_ma5, "state": state}


def _percentile(series, value):
    """value 在 series 中的分位（%）：<= value 的比例。series 可含 None/NaN。"""
    if value is None:
        return None
    vals = [float(x) for x in (series or [])
            if x is not None and not (isinstance(x, float) and math.isnan(x))]
    if not vals:
        return None
    return round(sum(1 for x in vals if x <= value) / len(vals) * 100, 2)


def compute_valuation(df: pd.DataFrame, hist_pe, hist_pb) -> dict:
    """PE/PB 中位数 + 近一年分位。"""
    if df is None or df.empty:
        return {"pe_median": None, "pb_median": None, "pe_pct_1y": None, "pb_pct_1y": None}
    pe = pd.to_numeric(df.get("peTTM"), errors="coerce")
    pb = pd.to_numeric(df.get("pbMRQ"), errors="coerce")
    pe_med = None if pe.dropna().empty else float(pe.median())
    pb_med = None if pb.dropna().empty else float(pb.median())
    return {"pe_median": pe_med, "pb_median": pb_med,
            "pe_pct_1y": _percentile(hist_pe, pe_med),
            "pb_pct_1y": _percentile(hist_pb, pb_med)}
```

```python
def compute_industry(df: pd.DataFrame, industry_map: dict, top_n: int = 5) -> dict:
    """按行业分组求平均涨跌幅，返回领涨/领跌 TOP n。industry_map: {code: industry}。"""
    if df is None or df.empty:
        return {"top": [], "bottom": []}
    d = df.copy()
    d["industry"] = d["code"].map(lambda c: (industry_map or {}).get(c, "") or "")
    d = d[d["industry"] != ""]
    d["pctChg"] = pd.to_numeric(d["pctChg"], errors="coerce")
    d = d[d["pctChg"].notna()]
    if d.empty:
        return {"top": [], "bottom": []}
    g = d.groupby("industry")["pctChg"].agg(["mean", "count"]).sort_values("mean", ascending=False)

    def _items(frame):
        return [{"industry": str(i), "avg_pct": round(float(r["mean"]), 2),
                 "count": int(r["count"])} for i, r in frame.iterrows()]

    return {"top": _items(g.head(top_n)), "bottom": _items(g.tail(top_n).iloc[::-1])}


def compute_hot(df: pd.DataFrame, name_map: dict, top_n: int = 10) -> dict:
    """涨/跌幅榜 TOP n（剔停牌），附股票名称。"""
    if df is None or df.empty:
        return {"gainers": [], "losers": []}
    st_num = pd.to_numeric(df.get("tradestatus"), errors="coerce")
    vol = pd.to_numeric(df.get("volume"), errors="coerce").fillna(0)
    live = df[(st_num == 1) & (vol > 0)].copy()
    live["pctChg"] = pd.to_numeric(live["pctChg"], errors="coerce")
    live = live[live["pctChg"].notna()]

    def _items(frame):
        return [{"code": str(r["code"]), "name": (name_map or {}).get(r["code"], ""),
                 "pctChg": round(float(r["pctChg"]), 2)} for _, r in frame.iterrows()]

    g = live.sort_values("pctChg", ascending=False)
    return {"gainers": _items(g.head(top_n)), "losers": _items(g.tail(top_n).iloc[::-1])}

# ---------- IO 层 ----------

def _parquet_glob(data_dir) -> str:
    return Path(data_dir).as_posix().rstrip("/") + "/*.parquet"


def _has_parquet(data_dir) -> bool:
    d = Path(data_dir)
    return d.is_dir() and any(d.glob("*.parquet"))


def _read_cross_section(data_dir, trade_date: str) -> pd.DataFrame:
    """读全市场当日的横截面（DuckDB 谓词下推）。无文件 → 空表。"""
    cols = "code,pctChg,tradestatus,volume,amount,turn,peTTM,pbMRQ,isST"
    if not _has_parquet(data_dir):
        return pd.DataFrame(columns=cols.split(","))
    import duckdb
    con = duckdb.connect()
    try:
        q = (f"SELECT {cols} FROM read_parquet('{_parquet_glob(data_dir)}') "
             f"WHERE date = '{trade_date}'")
        return con.execute(q).df()
    finally:
        con.close()


def _daily_amount_series(data_dir, trade_date: str, n: int = 6) -> list:
    """按日聚合的全市场成交额，返回 [(date, amount)] 降序（含当日）。"""
    if not _has_parquet(data_dir):
        return []
    import duckdb
    con = duckdb.connect()
    try:
        q = (f"SELECT date, SUM(amount) AS amount FROM "
             f"read_parquet('{_parquet_glob(data_dir)}') WHERE date <= '{trade_date}' "
             f"GROUP BY date ORDER BY date DESC LIMIT {n}")
        return [(str(d), float(a) if a is not None else 0.0) for d, a in con.execute(q).fetchall()]
    finally:
        con.close()


def _daily_median_series(data_dir, trade_date: str, col: str, days: int = 250) -> list:
    """按日的某列全市场中位数序列（近 days 日，含当日），用于算近一年分位。"""
    if not _has_parquet(data_dir):
        return []
    import duckdb
    con = duckdb.connect()
    try:
        q = (f"SELECT MEDIAN({col}) AS m FROM read_parquet('{_parquet_glob(data_dir)}') "
             f"WHERE date <= '{trade_date}' GROUP BY date ORDER BY date DESC LIMIT {days}")
        return [float(m) if m is not None else None for (m,) in con.execute(q).fetchall()]
    finally:
        con.close()
```

```python
# -*- coding: utf-8 -*-
"""报告渲染：把指标 dict 渲染为 HTML 邮件（Jinja2 模板）。"""
from pathlib import Path

from jinja2 import Environment, FileSystemLoader, select_autoescape

UP_COLOR = "#d60000"     # A股红涨
DOWN_COLOR = "#1a9c3c"   # A股绿跌
FLAT_COLOR = "#6a737d"

_TPL_DIR = Path(__file__).resolve().parents[1] / "templates"   # app/templates
_env = Environment(loader=FileSystemLoader(str(_TPL_DIR)),
                   autoescape=select_autoescape(["html", "xml"]))


def _pctcolor(v):
    try:
        v = float(v)
    except (TypeError, ValueError):
        return FLAT_COLOR
    if v > 0:
        return UP_COLOR
    if v < 0:
        return DOWN_COLOR
    return FLAT_COLOR


def _pct(v):
    if v is None:
        return "—"
    try:
        return f"{float(v):+.2f}%"
    except (TypeError, ValueError):
        return "—"


def _fmtnum(v, nd=2):
    if v is None:
        return "—"
    try:
        return f"{float(v):.{nd}f}"
    except (TypeError, ValueError):
        return "—"


def _yi2(v):
    if v is None:
        return "—"
    try:
        return f"{float(v) / 1e8:.1f}"
    except (TypeError, ValueError):
        return "—"


_env.filters["pctcolor"] = _pctcolor
_env.filters["pct"] = _pct
_env.filters["fmtnum"] = _fmtnum
_env.filters["yi2"] = _yi2


def render_report(metrics: dict, trade_date: str, subject_prefix: str = "") -> str:
    """渲染邮件 HTML。metrics 见 market_summary.build_summary。"""
    tpl = _env.get_template("daily_report.html.j2")
    return tpl.render(m=metrics, trade_date=trade_date, subject_prefix=subject_prefix)
```

```python
# -*- coding: utf-8 -*-
"""SMTP 发送：126 邮箱 SSL（465），认证用客户端授权码（由调用方从环境变量取得）。"""
import logging
import smtplib
from email.header import Header
from email.mime.text import MIMEText
from email.utils import formataddr

logger = logging.getLogger("app.services.email_sender")


def send_email(cfg, recipients, subject: str, html: str, password: str) -> None:
    """构造 HTML 邮件并发送。cfg 提供 smtp_* 属性；失败抛异常由上层记录。"""
    if not recipients:
        raise ValueError("收件人列表为空")
    from_addr = cfg.smtp_from or cfg.smtp_username
    msg = MIMEText(html, "html", "utf-8")
    msg["Subject"] = Header(subject, "utf-8")
    msg["From"] = formataddr((str(Header("量化平台", "utf-8")), from_addr))
    msg["To"] = ", ".join(recipients)

    cls = smtplib.SMTP_SSL if cfg.smtp_use_ssl else smtplib.SMTP
    with cls(cfg.smtp_host, cfg.smtp_port, timeout=20) as smtp:
        smtp.login(cfg.smtp_username, password)
        smtp.send_message(msg)
    logger.info("邮件已发送：%s → %s", from_addr, recipients)
```

```python
# -*- coding: utf-8 -*-
"""每日大盘汇总编排：就绪检查 → 指数补抓 → 汇总 → 渲染归档 → 发送。

- 最近交易日：由 ingest._trade_dates 钩子取（接线层已绑真实数据源）。
- 就绪条件：个股最新日期 == 交易日，且至少一只配置指数有该日数据。
- 指数补抓为 best-effort：缺失的指数跳过，不阻断；缺失板块渲染为「数据缺失」。
- 发送缺配置/缺授权码时记为未发送（不报错），状态仍为 success。
"""
import logging
import os
from datetime import date, datetime
from pathlib import Path

from sqlalchemy import func

from ..models import CronTask, StockFreshness
from . import email_sender, index_ingest, index_store, market_summary, report_render

logger = logging.getLogger("app.services.report_service")

REPORT_REF_NAME = "每日大盘汇总"


def resolve_trade_date(today: str | None = None, lookback_days: int = 15):
    """最近交易日（<= today）：走 ingest._trade_dates 钩子；取不到返回 None。"""
    from . import ingest
    today = today or date.today().isoformat()
    start = (date.fromisoformat(today) - __import__("datetime").timedelta(days=lookback_days)).isoformat()
    days = ingest._trade_dates(start, today)
    return days[-1] if days else None


def _stock_latest(db, data_dir) -> str | None:
    """个股最新日期：优先元库断点表最大值，其次扫描目录内文件 date 列。"""
    try:
        row = db.query(func.max(StockFreshness.latest_date)).scalar()
    except Exception:  # noqa: BLE001
        row = None
    if row is not None:
        return row.isoformat() if hasattr(row, "isoformat") else str(row)
    d = Path(data_dir)
    if not d.is_dir():
        return None
    best = None
    for p in sorted(d.glob("*.parquet"))[:200]:
        try:
            df = __import__("pandas").read_parquet(p, columns=["date"])
            if not df.empty:
                v = str(df["date"].max())
                best = v if best is None or v > best else best
        except Exception:  # noqa: BLE001
            continue
    return best


def readiness_check(cfg, db, trade_date: str) -> dict:
    """返回 {ready, stock_latest, index_ready, reasons}。"""
    s_latest = _stock_latest(db, cfg.data_dir)
    idx_ready = [c for c in cfg.report_index_codes
                 if index_store.latest_date(cfg.index_dir, c) == trade_date]
    reasons = []
    if s_latest != trade_date:
        reasons.append(f"个股最新日期 {s_latest or '无'} != 交易日 {trade_date}")
    if not idx_ready:
        reasons.append(f"无指数数据落在交易日 {trade_date}")
    return {"ready": not reasons, "stock_latest": s_latest,
            "index_ready": idx_ready, "reasons": reasons}


def _stock_maps(db):
    """从 stock 表取 {code: industry} 与 {code: name}。表空/异常返回空 dict。"""
    from ..models import Stock
    try:
        rows = db.query(Stock).all()
    except Exception:  # noqa: BLE001
        return {}, {}
    return ({r.code: (r.industry or "") for r in rows},
            {r.code: (r.name or "") for r in rows})


def generate(cfg, db, trade_date: str) -> dict:
    """汇总 + 渲染 + 归档，返回 {trade_date, metrics, html, html_path}。"""
    industry_map, name_map = _stock_maps(db)
    metrics = market_summary.build_summary(
        cfg.data_dir, cfg.index_dir, trade_date,
        cfg.report_index_codes, industry_map, name_map)
    html = report_render.render_report(metrics, trade_date, cfg.report_subject_prefix)
    out_dir = Path(cfg.output_dir) / "reports"
    out_dir.mkdir(parents=True, exist_ok=True)
    html_path = out_dir / f"{trade_date}.html"
    html_path.write_text(html, encoding="utf-8")
    return {"trade_date": trade_date, "metrics": metrics,
            "html": html, "html_path": str(html_path)}


def _send(cfg, trade_date: str, html: str, log) -> dict:
    """按配置发送；缺配置/失败记原因，不抛异常。"""
    if not cfg.report_recipients:
        log("未配置收件人(report.recipients)，跳过发送")
        return {"sent": False, "reason": "未配置 report.recipients"}
    password = os.environ.get(cfg.smtp_password_env, "")
    if not (cfg.smtp_username and password):
        log(f"缺 smtp.username 或环境变量 {cfg.smtp_password_env}，跳过发送")
        return {"sent": False, "reason": f"缺 smtp.username 或环境变量 {cfg.smtp_password_env}"}
    subject = f"{cfg.report_subject_prefix} | {trade_date}"
    try:
        email_sender.send_email(cfg, cfg.report_recipients, subject, html, password)
        log(f"邮件已发送至 {cfg.report_recipients}")
        return {"sent": True, "recipients": list(cfg.report_recipients)}
    except Exception as exc:  # noqa: BLE001
        log(f"邮件发送失败：{exc}")
        return {"sent": False, "reason": str(exc)}


def run_daily_report(cfg, db, log=None, send: bool = True, today: str | None = None) -> dict:
    """完整流程。返回 dict（含 status/trade_date/reason/metrics/html_path/email/output_files）。"""
    def _log(m: str) -> None:
        if log:
            log(m)
        logger.info(m)

    trade_date = resolve_trade_date(today)
    if not trade_date:
        _log("无法确定最近交易日（数据源未就绪），跳过")
        return {"status": "skipped", "trade_date": None,
                "reason": "无法确定最近交易日", "output_files": []}
    _log(f"最近交易日 = {trade_date}")

    try:
        r = index_ingest.run_index_fetch(cfg.index_dir, cfg.report_index_codes,
                                         trade_date, log=_log)
        _log(f"指数抓取：新增 {len(r['fetched'])}，跳过 {len(r['skipped'])}")
    except Exception as exc:  # noqa: BLE001
        _log(f"指数抓取异常（不阻断）：{exc}")

    ready = readiness_check(cfg, db, trade_date)
    if not ready["ready"]:
        reason = "；".join(ready["reasons"])
        _log(f"数据未就绪，跳过发送：{reason}")
        return {"status": "skipped", "trade_date": trade_date, "reason": reason,
                "stock_latest": ready["stock_latest"],
                "index_ready": ready["index_ready"], "output_files": []}

    gen = generate(cfg, db, trade_date)
    _log(f"已生成并归档：{gen['html_path']}")

    email_info = _send(cfg, trade_date, gen["html"], _log) if send else {"sent": False, "reason": "预览模式未发送"}
    return {"status": "success", "trade_date": trade_date, "reason": "",
            "metrics": gen["metrics"], "html_path": gen["html_path"],
            "email": email_info, "output_files": [gen["html_path"]]}

        return

                    cron_expr=cfg.report_cron, params={}, enabled=cfg.report_enabled))
```

```python
# -*- coding: utf-8 -*-
"""报告 API：立即生成并发送 / 预览 HTML（不发信）/ 最近一次记录。"""
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session

from ..config import load_config
from ..database import get_db
from ..models import TaskRun
from ..services import report_service
from ..services.task_service import get_service

router = APIRouter(prefix="/api/reports", tags=["reports"])


@router.get("/daily/preview")
def preview(db: Session = Depends(get_db)):
    """渲染当日汇总 HTML（不发送、不落库；不含指数补抓等写操作）。"""
    cfg = load_config()
    trade_date = report_service.resolve_trade_date()
    if not trade_date:
        raise HTTPException(409, "无法确定最近交易日（数据源未就绪）")
    gen = report_service.generate(cfg, db, trade_date)
    return {"trade_date": trade_date, "html": gen["html"]}


@router.post("/daily/run")
def run(db: Session = Depends(get_db)):
    """立即生成并发送当日汇总（入队，走任务日志/状态）。"""
    svc = get_service()
    task = svc.enqueue_report(db)
    return {"task_id": task.id, "kind": task.kind}


@router.get("/daily/latest")
def latest(db: Session = Depends(get_db)):
    """最近一次 report 任务的摘要（含归档 HTML 路径与状态）。"""
    row = (db.query(TaskRun).filter(TaskRun.kind == "report")
           .order_by(TaskRun.id.desc()).first())
    if row is None:
        return {"task_id": None, "status": "", "result_summary": {}, "output_files": []}
    return {"task_id": row.id, "status": row.status,
            "result_summary": row.result_summary or {},
            "output_files": row.output_files or []}
```
